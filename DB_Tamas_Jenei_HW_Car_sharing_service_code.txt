Table persons {
  person_id int [
    pk,
    increment,
    not null,
    note: 'PK, SuK, SK' ]
  first_name varchar(50) [
    not null  ]
  last_name varchar(50) [
    not null  ]
  birth_date date
  personal_id varchar(32) [
    unique,
    note: 'AK, NK, CK, SK'  ]
  email varchar(100) [
    not null,
    unique,
    note: 'AK, NK, CK, SK'  ]}
